1. callback function:
                       A callback is a function passed as an argument to another function.
                       It runs after the main function finishes some work (usually async work).
                       It helps js handle tasks in order without blocking other code .

            syntax:
                      function mainFunction(callback)
                      {
                        //codecallback();  
                      } //calling the callback.


Example:
         function greet(name, callback)
{
    console.log("inside  functyion");
    console.log("hello "+ name);//main work 
    callback();
}

function aftergreeting()
{
    console.log("callback");
}

greet("john",aftergreeting);
