map():
       map() creates a new array 
       it does not change the original array 
       it is mainly used when you want to transform all items 

    Example:
          
let a =[1,2,3];
             
            // 1   1 *2=2 
            // 2   2 *2=4
            // 3   3 *2=6
let b = a.map(x => x * 2);

console.log(a);//1,2,3
console.log(b);//2,4,6   
-----------------------------------------------------------------------

reduce()
         total/sum
   Example:
    let a =[1,2,3,4,5];
                    // 0   1       0+1=1
                 //    1   2       1+2=3
                 //    3    3      3+3 =6
                 //    6    4      6+4=10
                 //    10   5       10+5=15
let total = a.reduce((acc,curr)=> acc+curr , 0);

console.log(total);
